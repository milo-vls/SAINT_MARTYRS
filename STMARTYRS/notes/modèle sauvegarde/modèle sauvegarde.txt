
//convertit une chaîne en fichier
function stringToFichier(nom_fichier, _string)
{

var _buffer = buffer_create(string_byte_length(_string) + 1, buffer_fixed, 1);
buffer_write(_buffer, buffer_string, _string);
buffer_save(_buffer, nom_fichier);
buffer_delete(_buffer);



}
//conversion de fichier en json
function fichierToJson(nom_fichier)
{
	var _buffer = buffer_load(nom_fichier);
	var _string = buffer_read(_buffer, buffer_string);
	buffer_delete(_buffer);
	var _json = json_parse(_string);
	return _json

}
